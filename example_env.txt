# Для запуску додатку необхідно:
# 1) Зазначити змінні які знаходяться нижче
# 2) Змінити назву файлу на [.env] - без []

### Необхідно вказати назву БД, на приклад DB_NAME=MyAppDbName ###
DB_NAME=MyAppDbName

### Необхідно вказати логін користувача в БД, на приклад DB_USER=postgres для PostgreSQL СУБД ###
DB_USER=postgres

### Необхідно вказати пароль користувача в БД, на приклад DB_PASSWORD=MyPassword ###
DB_PASSWORD=MyPassword
